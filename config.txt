# RSE-Fixes - settings
# Change them in Esc > MODS if you use Mod Menu, or edit this file with the game closed.

# ---- Debug ----
# true = automatic reports and log lines while you work on the building bug (see below).
# false = the mod stays quiet and does nothing until you type fixes_buildings / fixes_players
# or use the Mod Menu button.
Debug = false

# ---- Fix: falling through buildings after a join or teleport (server or host only) ----
# true = when a player from another machine joins or teleports, the server holds them in place until
# the building under them has collision on the server, puts them on top of it and lets them go.
# Arriving on the ground: no hold. Needs the mod on the server (dedicated) or the host's game.
HoldArrivals = false

# Longest hold in seconds before letting them go where they are (2 to 30).
HoldSeconds = 10

# ---- Building diagnostics ----
# With Debug on: true = after a teleport, a join or a respawn, and whenever your footing turns unstable (walking and
# falling flicker, upward snaps), write a short report on the buildings around you to UE4SS.log.
AutoDiagnose = true

# Metres around your character to inspect (5 to 60).
Radius = 15

# Metres moved within a quarter of a second that count as a teleport (20 to 500).
TeleportDistance = 50
